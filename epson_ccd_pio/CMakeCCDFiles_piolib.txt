# Set SOURCES variable
cmake_minimum_required(VERSION 3.16)

project(ccd_control LANGUAGES C)

message("Epson ccd pio included")

add_custom_command(
  OUTPUT ${CMAKE_CURRENT_LIST_DIR}/pio_ccd.h
  COMMAND pioasm pio_ccd.pio pio_ccd.pio.h
  WORKING_DIRECTORY ${CMAKE_CURRENT_LIST_DIR}
  DEPENDS ${CMAKE_CURRENT_LIST_DIR}/pio_ccd.pio
)

if(${PIOLIB})
  message("Piolib on")
  add_compile_definitions(PIOLIB)
  set(PIOPATH /home/pi/git/utils/piolib/)
  link_directories(${PIOPATH})
  include_directories(/home/pi/git/utils/piolib/include/)
endif()

set(CCD_SOURCES
  ${CMAKE_CURRENT_LIST_DIR}/ccd_commands.c ${CMAKE_CURRENT_LIST_DIR}/ccd_itf_common.c
)

# Set HEADERS_PUBLIC variable (public headers, included in the library)
set(CCD_HEADERS_PUBLIC
  ${CMAKE_CURRENT_LIST_DIR}/ccd_commands.h ${CMAKE_CURRENT_LIST_DIR}/ccd_itf_common.h ${CMAKE_CURRENT_LIST_DIR}/pio_ccd.h
)
