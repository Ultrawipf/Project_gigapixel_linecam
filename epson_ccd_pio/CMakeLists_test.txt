# Set SOURCES variable
cmake_minimum_required(VERSION 3.16)

project(ccd_control LANGUAGES C)

message("Epson ccd pio included")

add_custom_command(
  OUTPUT ${CMAKE_CURRENT_SOURCE_DIR}/pio_ccd.h
  COMMAND pioasm pio_ccd.pio pio_ccd.h
  WORKING_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}
  DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/pio_ccd.pio
)

if(${PIOLIB})
  add_compile_definitions(PIOLIB)
  set(PIOPATH /home/pi/git/utils/piolib/)
  link_directories(${PIOPATH})
  include_directories(/home/pi/git/utils/piolib/include/)
endif()

set(SOURCES
  ccd_commands.c ccd_itf_common.c
)

# Set HEADERS_PUBLIC variable (public headers, included in the library)
set(HEADERS_PUBLIC
  ccd_commands.h ccd_itf_common.h
)

add_library(ccd_control ${SOURCES} ${HEADERS_PUBLIC})

if(${PIOLIB})
  message("Epson ccd link piolib")
  target_link_libraries(ccd_control pio)
endif()
